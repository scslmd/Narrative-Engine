"""Parametric parity tests between legacy and /v1 endpoints."""
from __future__ import annotations

import csv
import json
from pathlib import Path

import pytest
from fastapi.testclient import TestClient

from app.main import build_app

# Load route pairs from the mapping spec generated by PR-A
MAPPING_CSV = Path(__file__).parent.parent / "docs" / "api-migration" / "route_classification.csv"


def _load_route_pairs() -> list[dict]:
    """Load only routes that are explicitly mapped to an existing /v1 counterpart."""
    pairs = []
    with open(MAPPING_CSV, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            cls = row["classification"]
            # Only include dual routes; PR-A marks legacy routes without counterparts as gaps.
            if cls == "dual":
                legacy_path = row["path"]
                v1_path = "/v1" + legacy_path
                method = row["method"].lower()
                pairs.append({
                    "method": method,
                    "legacy_path": legacy_path,
                    "v1_path": v1_path,
                    "tags": row.get("tags", ""),
                })
    return pairs


ROUTE_PAIRS = _load_route_pairs()

# Routes that require authentication — skip parity for auth-gated routes
# since TestClient won't have API key configured by default.
AUTH_GATED_TAGS = {"projects"}  # /projects/import-story etc. require API key


@pytest.mark.parametrize(
    "route_pair",
    ROUTE_PAIRS,
    ids=lambda rp: f"{rp['method'].upper()} {rp['legacy_path']}",
)
def test_route_parity(route_pair: dict, tmp_path) -> None:
    """Assert legacy and /v1 endpoints return identical status codes for GET requests."""
    method = route_pair["method"]
    legacy_path = route_pair["legacy_path"]
    v1_path = route_pair["v1_path"]
    tags = route_pair.get("tags", "")

    # Skip POST/PUT/PATCH/DELETE routes that require request bodies
    if method != "get":
        pytest.skip(f"Non-GET route: {method.upper()} {legacy_path}")

    # Skip auth-gated routes
    tag_list = [t.strip() for t in tags.split(";") if t.strip()]
    if any(t in AUTH_GATED_TAGS for t in tag_list):
        pytest.skip(f"Auth-gated route: {legacy_path}")

    client = TestClient(build_app())

    legacy_resp = client.get(legacy_path)
    v1_resp = client.get(v1_path)

    assert (
        legacy_resp.status_code == v1_resp.status_code
    ), f"Status mismatch for {legacy_path}: legacy={legacy_resp.status_code}, v1={v1_resp.status_code}"

    # Compare response bodies for 2xx responses
    if 200 <= legacy_resp.status_code < 300:
        try:
            legacy_body = legacy_resp.json()
            v1_body = v1_resp.json()
            assert legacy_body == v1_body, (
                f"Body mismatch for {legacy_path}: "
                f"legacy keys={set(str(k) for k in (legacy_body if isinstance(legacy_body, dict) else []))}, "
                f"v1 keys={set(str(k) for k in (v1_body if isinstance(v1_body, dict) else []))}"
            )
        except json.JSONDecodeError:
            # Non-JSON responses (e.g., HTML, text) — compare raw text
            assert legacy_resp.text == v1_resp.text, f"Text body mismatch for {legacy_path}"


def test_all_get_routes_have_parity_coverage() -> None:
    """Ensure the mapping table covers all expected dual routes."""
    get_pairs = [rp for rp in ROUTE_PAIRS if rp["method"] == "get"]
    # We expect at least some GET routes to be covered
    assert len(get_pairs) > 0, "No GET route pairs found in mapping — check classification CSV"
```

from __future__ import annotations

import pytest
from fastapi.testclient import TestClient

from app.main import build_app


class TestErrorSemanticsNotFound:
    """404 parity: both legacy and /v1 return 404 for missing resources."""

    def test_backup_404_parity(self, tmp_path) -> None:
        client = TestClient(build_app())
        legacy = client.get("/backup/latest")
        v1 = client.get("/v1/backup/latest")
        # Both should return 200 (empty) or consistent behavior
        assert legacy.status_code == v1.status_code

    def test_project_404_legacy(self, tmp_path) -> None:
        client = TestClient(build_app())
        resp = client.get("/projects/nonexistent-uuid")
        assert resp.status_code == 404

    def test_project_404_v1(self, tmp_path) -> None:
        client = TestClient(build_app())
        resp = client.get("/v1/projects/nonexistent-uuid")
        assert resp.status_code == 404


class TestErrorSemanticsAuth:
    """401 parity: both legacy and /v1 return 401 when API key is required but missing."""

    def test_import_story_401_without_key(self, tmp_path) -> None:
        """Both /projects/import-story paths require API key."""
        import app.settings as settings_module
        original_key = settings_module.settings.api_key
        try:
            settings_module.settings.api_key = "test-key-for-parity"
            client = TestClient(build_app())

            payload = {"story_text": "x" * 100, "project_name": "test"}
            resp = client.post("/projects/import-story", json=payload)
            assert resp.status_code == 401, f"Expected 401, got {resp.status_code}"
            resp_v1 = client.post("/v1/projects/import-story", json=payload)
            assert resp_v1.status_code == 401, f"Expected 401, got {resp_v1.status_code}"
        finally:
            settings_module.settings.api_key = original_key

    def test_auth_keys_403_parity_for_insufficient_permissions(self, tmp_path) -> None:
        """Both auth route variants return the same status/body for forbidden calls."""
        client = TestClient(build_app())
        headers = {"X-API-Key": "insufficient-scope-key"}
        legacy = client.get("/auth/keys", headers=headers)
        v1 = client.get("/v1/auth/keys", headers=headers)
        assert legacy.status_code == v1.status_code == 403
        assert legacy.json().get("detail") == v1.json().get("detail")


class TestErrorSemanticsValidation:
    """400 parity: both legacy and /v1 return 400 for invalid input."""

    def test_auth_create_empty_name_parity(self, tmp_path) -> None:
        client = TestClient(build_app())
        payload = {"name": "", "permissions": ["read"]}
        legacy = client.post("/auth/keys", json=payload)
        v1 = client.post("/v1/auth/keys", json=payload)
        assert legacy.status_code == v1.status_code == 400

    def test_auth_invalid_permission_parity(self, tmp_path) -> None:
        client = TestClient(build_app())
        payload = {"name": "test", "permissions": ["invalid_perm"]}
        legacy = client.post("/auth/keys", json=payload)
        v1 = client.post("/v1/auth/keys", json=payload)
        assert legacy.status_code == v1.status_code == 400


class TestErrorSemanticsConflict:
    """409 parity: both legacy and /v1 return 409 for conflict scenarios."""

    def test_projects_create_conflict_parity(self, tmp_path) -> None:
        """Creating duplicate project identifiers should return 409 on both routes."""
        client = TestClient(build_app())
        payload = {"project_name": "parity-conflict-project", "genre": "sci-fi"}
        first = client.post("/projects/create", json=payload)
        first_v1 = client.post("/v1/projects/create", json=payload)
        assert first.status_code in (200, 201)
        assert first_v1.status_code in (200, 201)

        legacy = client.post("/projects/create", json=payload)
        v1 = client.post("/v1/projects/create", json=payload)
        assert legacy.status_code == v1.status_code == 409
        assert legacy.json().get("detail") == v1.json().get("detail")

"""Middleware that emits structured telemetry for legacy route access."""
from __future__ import annotations

import json
import logging
import time
from pathlib import Path
from typing import Awaitable, Callable

from fastapi import Request, Response

logger = logging.getLogger(__name__)

# Routes exempted from legacy telemetry (health, internal)
EXEMPT_PREFIXES = ("/health", "/docs", "/redoc", "/openapi.json", "/assets", "/static")


async def legacy_route_telemetry_middleware(
    request: Request,
    call_next: Callable[[Request], Awaitable[Response]],
) -> Response:
    """Emit a structured log event when a legacy (non-/v1) route is accessed.

    Events are written to the structured log file as JSON lines with the
    event type `legacy_route_hit`. Downstream tooling can query these events
    to determine when legacy routes have zero usage for N consecutive days.
    """
    path = request.url.path

    # Skip exempted paths
    if any(path.startswith(prefix) for prefix in EXEMPT_PREFIXES):
        return await call_next(request)

    # Only log non-/v1 routes that have a /v1 counterpart
    is_legacy = not path.startswith("/v1/") and path not in ("/", "/role-model-checker-ui", "/vite.svg")

    if is_legacy:
        event = {
            "event": "legacy_route_hit",
            "timestamp": time.time(),
            "method": request.method,
            "path": path,
            "client_host": request.client.host if request.client else "unknown",
            "user_agent": request.headers.get("user-agent", ""),
        }
        logger.info(json.dumps(event))

    response = await call_next(request)
    return response
```

## Zero-Usage Check

To verify zero legacy hits for the past N days:

```python
import json
from pathlib import Path
from datetime import datetime, timezone, timedelta

log_path = Path("data/.structured_log")  # or settings.structured_log_filename
cutoff = (datetime.now(timezone.utc) - timedelta(days=14)).timestamp()

hits_after_cutoff = 0
with open(log_path, "r", encoding="utf-8") as f:
    for line in f:
        try:
            event = json.loads(line.strip())
            if event.get("event") == "legacy_route_hit" and event["timestamp"] > cutoff:
                hits_after_cutoff += 1
        except (json.JSONDecodeError, KeyError):
            continue

print(f"Legacy hits in last 14 days: {hits_after_cutoff}")
if hits_after_cutoff == 0:
    print("PASS: Safe to remove legacy routes")
else:
    print(f"FAIL: {hits_after_cutoff} legacy hits detected — wait longer")

# API Migration Cutover Gate

## Purpose

Define deterministic, measurable criteria for removing legacy (unversioned) API routes and completing the `/v1` canonical migration.

## Gate Criteria (ALL must pass)

### 1. Parity Tests Green

```bash
python -m pytest -q -p no:cacheprovider tests/test_api_route_parity.py
```

- **Pass:** All parametrized route pairs return identical status codes and response bodies.
- **Fail:** Any assertion failure or unexpected error.

### 2. Error Semantics Tests Green

```bash
python -m pytest -q -p no:cacheprovider tests/test_api_error_semantics.py
```

- **Pass:** All 401/403/404/409 paths are identical between legacy and `/v1`.
- **Fail:** Any status code or error body mismatch.

### 3. Frontend Canonical-Only Check Green

```bash
rg -n "api\\.(get|post|patch|put|delete)\\((`|'|\")/(?!v1)" frontend/src/services frontend/src/lib -S
rg -n "API_BASE|fetch\\(|axios\\.(get|post|patch|put|delete)\\((`|'|\")/(?!v1)" frontend/src/services frontend/src/lib -S

- **Pass:** Both commands return no matches (exit code 1).
- **Fail:** Any legacy endpoint string found in production service layer.

### 4. Legacy Hits = 0 for 14 Days

Run the zero-usage query from `docs/api-migration/legacy_usage_query.md`:

```bash
python scripts/check_legacy_usage.py --days 14

- **Pass:** Output shows `Legacy hits in last 14 days: 0`.
- **Fail:** Any hits detected.

### 5. Full Verification Suite Green

Run the complete merge-readiness validation (see PR-E Task 022).

1. Run criteria 1-3 in parallel (independent checks).
2. If all pass, run criterion 4 (telemetry check).
3. If telemetry passes, proceed to PR-E (legacy removal).
4. If any criterion fails, document the failure and re-evaluate after remediation.

## Escalation

If legacy hits persist beyond 60 days despite frontend migration:
1. Audit all external integrations for hardcoded legacy endpoints.
2. Send deprecation notice to integration owners via `Link` header URL.
3. Extend sunset date by 30 days with documented justification.
